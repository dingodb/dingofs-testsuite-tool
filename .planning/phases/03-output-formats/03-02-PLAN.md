```bash
fio_run() {
    local config
    config=$(get_scenario_path fio "$SCENARIO")

    echo "Running fio with config: $config"
    echo "  Mount: $MOUNT"
    echo "  Output: $OUTPUT"

    mkdir -p "$OUTPUT"

    # Capture both stdout and stderr to raw file while also letting fio write JSON
    "$FIO_BIN" "$config" "--output=$OUTPUT/fio.json" "--output-format=json" \
        2>&1 | tee "$OUTPUT/fio.raw"

    # If directory override needed
    if [[ -d "$MOUNT" ]]; then
        # Re-run with directory if not already in config (simplified for this example)
        :
    fi

    # Generate reports
    echo "Generating reports..."
    python3 /scripts/generate_report.py --tool fio --output-dir "$OUTPUT"

    return ${PIPESTATUS[0]}
}
```

In entrypoint.sh, modify the vdbench_run() function:

1. Capture raw vdbench output to $OUTPUT/vdbench.raw:
   - vdbench outputs to directory specified by -o flag
   - Use tee to capture the console output: ./vdbench ... 2>&1 | tee "$OUTPUT/vdbench.raw"
   - vdbench also creates output files in the -o directory

2. After vdbench completes, call report generation:
   python3 /scripts/generate_report.py --tool vdbench --output-dir "$OUTPUT"

3. The vdbench output files will be in $OUTPUT/ directory (vdbench creates parser_current and other files)

Example modification:
```bash
vdbench_run() {
    local config
    config=$(get_scenario_path vdbench "$SCENARIO")

    echo "Running vdbench with config: $config"
    echo "  Mount: $MOUNT"
    echo "  Output: $OUTPUT"

    mkdir -p "$OUTPUT"

    cd "$VDBENCH_DIR"
    "./vdbench" -f "$config" -o "$OUTPUT" 2>&1 | tee "$OUTPUT/vdbench.raw"
    local vdbench_exit=${PIPESTATUS[0]}

    # Generate reports
    echo "Generating reports..."
    python3 /scripts/generate_report.py --tool vdbench --output-dir "$OUTPUT"

    return $vdbench_exit
}
```
