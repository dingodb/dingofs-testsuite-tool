## Architecture Decision

Per research (Option A: container-side notification), the combined notification function lives inside entrypoint.sh and is called by smoke_run() after generate_smoke_summary(). The dtt wrapper cmd_smoke() simply passes WECHAT/EMAIL/WEBHOOK_URL/EMAIL_TO env vars to the container. This avoids host-side JSON parsing and keeps all notification logic in one place.

  **Step 1: Add `send_smoke_notification()` function definition**

  Insert the function between `generate_smoke_summary()` (ends around line 2029) and `smoke_run()` (starts at line 2032). The function should be placed at a clean line boundary after the `}` closing `generate_smoke_summary`.

  ```bash
  # Send combined smoke notification (WeChat + Email) with all three tools' results.
  # Arguments: $1 = smoke_base directory path; $2 = aggregate exit code; $3 = start timestamp (seconds since epoch).
  # Reads global SMOKE_PJD_*, SMOKE_LTP_*, SMOKE_MDT_PASS variables set by parse/validate functions.
  send_smoke_notification() {
      local smoke_base="$1"
      local aggregate_exit="$2"
      local start_ts="$3"

      # Calculate total duration
      local end_ts=$(date +%s)
      local duration_sec=$((end_ts - start_ts))
      local duration_str
      duration_str=$(printf '%dm%ds' $((duration_sec/60)) $((duration_sec%60)))

      # Determine aggregate status
      local agg_status="SUCCESS"
      if [[ $aggregate_exit -ne 0 ]]; then
          agg_status="FAIL"
      fi

      # Determine per-tool statuses
      local pjd_status="PASS"
      if [[ ${SMOKE_PJD_FAIL:-0} -ne 0 ]] || [[ ${SMOKE_PJD_TOTAL:-0} -eq 0 ]]; then
          pjd_status="FAIL"
      fi

      local mdt_status="PASS"
      if [[ ${SMOKE_MDT_PASS:-0} -ne 1 ]]; then
          mdt_status="FAIL"
      fi

      local ltp_status="PASS"
      if [[ ${SMOKE_LTP_TIMEOUT:-0} -eq 1 ]]; then
          ltp_status="TIMEOUT"
      elif [[ ${SMOKE_LTP_FAIL:-0} -ne 0 ]] || [[ ${SMOKE_LTP_TOTAL:-0} -eq 0 ]]; then
          ltp_status="FAIL"
      fi

      # Build combined details string for notification functions
      # Format: "pjdtest[PASS] pass:5 fail:0 skip:2 total:7; mdtest[PASS]; ltp[PASS] pass:12 fail:0 skip:3 total:15"
      local details="pjdtest[${pjd_status}] pass:${SMOKE_PJD_PASS:-0} fail:${SMOKE_PJD_FAIL:-0} skip:${SMOKE_PJD_SKIP:-0} total:${SMOKE_PJD_TOTAL:-0}; mdtest[${mdt_status}]; ltp[${ltp_status}] pass:${SMOKE_LTP_PASS:-0} fail:${SMOKE_LTP_FAIL:-0} skip:${SMOKE_LTP_SKIP:-0} total:${SMOKE_LTP_TOTAL:-0}"

      echo ""
      echo "[notify] Sending combined smoke notification (WeChat: $WECHAT_ENABLED, Email: $EMAIL_ENABLED)..."

      # Send WeChat notification (function sourced from notify.sh)
      send_wechat_notification "smoke" "smoke" "$agg_status" "$duration_str" "$details"

      # Send Email notification (function sourced from notify.sh)
      send_email_notification "smoke" "smoke" "$agg_status" "$duration_str" "$details"

      echo "[notify] Combined smoke notification sent."
  }
  ```

  ```bash
  cmd_smoke() {
      local image=$(load_config image)
      local config_testdir=$(load_config testdir)
      local config_output=$(load_config output)
      local config_mdsaddr=$(load_config mdsaddr)
      local runtime=$(load_config runtime)

      # Handle --help before config checks
      if [[ "$1" == "--help" ]] || [[ "$1" == "-h" ]]; then
          show_smoke_help
          return 0
      fi

      # Parse flags: only --wechat, --email, --debug, --help
      local wechat=""
      local email_to=""
      local debug=""
      while [[ $# -gt 0 ]]; do
          case "$1" in
              --wechat)
                  wechat="yes"
                  shift
                  ;;
              --email)
                  email_to="yes"
                  shift
                  ;;
              --debug)
                  debug="yes"
                  shift
                  ;;
              --help|-h)
                  show_smoke_help
                  return 0
                  ;;
              *)
                  echo "Error: Unknown option '$1'"
                  echo "Run 'dtt smoke --help' for usage information"
                  return 1
                  ;;
          esac
      done

      # Validate image is configured
      if [[ -z "$image" ]]; then
          echo "Error: No Docker image configured."
          echo "Please run 'dtt config set image <image>' first."
          return 1
      fi

      # Validate test directory
      local mount="${config_testdir}"
      if [[ -z "$mount" ]]; then
          echo "Error: Test directory not set."
          echo "Please run 'dtt config set testdir <directory>' first."
          return 1
      fi
      if [[ ! -d "$mount" ]]; then
          echo "Error: Mount directory '$mount' does not exist"
          return 1
      fi

      # Validate/create output directory
      local output="${config_output}"
      if [[ -z "$output" ]]; then
          echo "Error: Output directory not set."
          echo "Please run 'dtt config set output <directory>' first."
          return 1
      fi
      mkdir -p "$output"

      # Build docker/podman command (omit --rm in debug mode)
      local rm_flag="--rm"
      if [[ -n "$debug" ]]; then
          rm_flag=""
      fi

      local docker_cmd=(
          $runtime run $rm_flag --privileged
          -v "${mount}:/data"
          -v "${output}:/output"
      )

      # Pass MDS address if configured (needed for integration test context)
      if [[ -n "$config_mdsaddr" ]]; then
          docker_cmd+=("-e" "MDSADDR=$config_mdsaddr")
          # Add --add-host entries for MDS IPs
          local -a added_ips=()
          while IFS=',' read -ra ADDRS; do
              for addr in "${ADDRS[@]}"; do
                  addr=$(echo "$addr" | xargs)
                  [[ -n "$addr" ]] || continue
                  local ip="${addr%:*}"
                  local already_added=false
                  for existing in "${added_ips[@]}"; do
                      [[ "$existing" == "$ip" ]] && already_added=true && break
                  done
                  if [[ "$already_added" == "false" ]]; then
                      docker_cmd+=("--add-host" "mds:${ip}")
                      added_ips+=("$ip")
                  fi
              done
          done <<< "$config_mdsaddr"
      fi

      # WeChat notification env vars
      if [[ -n "$wechat" ]]; then
          local webhook_url=$(load_config webhook_url)
          docker_cmd+=("-e" "WECHAT=yes")
          docker_cmd+=("-e" "WEBHOOK_URL=$webhook_url")
      fi

      # Email notification env vars
      if [[ -n "$email_to" ]]; then
          local email_addr=$(load_config email)
          docker_cmd+=("-e" "EMAIL=yes")
          docker_cmd+=("-e" "EMAIL_TO=$email_addr")
      fi

      # Print execution info
      echo "Starting smoke test suite..."
      echo "  Image:       $image"
      echo "  Test dir:    $mount -> /data"
      echo "  Output dir:  $output -> /output"
      if [[ -n "$wechat" ]]; then
          echo "  WeChat:      enabled"
      fi
      if [[ -n "$email_to" ]]; then
          echo "  Email:       enabled"
      fi
      if [[ -n "$debug" ]]; then
          echo "  Debug:       container will be preserved"
      fi
      echo ""

      # Execute: run container with -t smoke, mounting test/output dirs
      # Entrypoint: /entrypoint.sh -t smoke -m /data -o /output
      "${docker_cmd[@]}" "$image" /entrypoint.sh -t smoke -m /data -o /output
  }
  ```
