# Base fio pattern:
# [global]
# name=<name>
# description=<description>
# rw=<randread|randwrite|read|write>
# bs=<blocksize>
# ioengine=libaio
# iodepth=1
# direct=<0|1>
# numjobs=<1|8|16|32>
# runtime=60
# time_based=1
# directory=/data
# size=1G
# group_reporting=1
# [job1]

# For random read/write, also include: norandommap=1

# Naming convention: {rw}_{direct}d_{bs}_{numjobs}j.fio
# Examples: rand_read_0d_128k_1j.fio, seq_write_1d_4m_32j.fio

# For bs in filenames use: 128k, 1m, 4m (lowercase, no space)
</context>

    Create a bash script at scenarios/fio/gen_fio_scenarios.sh that generates all 96 fio scenario files.

    The script should:
    1. Define a function generate_fio_scenario() that creates fio files
    2. Loop through all combinations:
       - rw_types: rand_read rand_write seq_read seq_write
       - direct: 0 1
       - bs_values: 128k 1m 4m
       - numjobs: 1 8 16 32
    3. Generate each file with proper fio syntax

    For rw mapping:
    - rand_read -> rw=randread
    - rand_write -> rw=randwrite
    - seq_read -> rw=read
    - seq_write -> rw=write

    For all scenarios (both sequential and random):
    - ioengine=libaio
    - iodepth=1 (FIXED - not 4 or 32)
    - runtime=60
    - time_based=1
    - directory=/data
    - size=1G
    - group_reporting=1

    For random scenarios only (rand_read, rand_write):
    - Include norandommap=1

    For sequential scenarios only (seq_read, seq_write):
    - Do NOT include norandommap

    Make the script executable with chmod +x.

    Template:
    #!/bin/bash
    # Generator script for 96 fio scenarios
    # 4 types x 2 direct x 3 bs x 4 numjobs = 96 files

    SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
    SCENARIOS_DIR="$SCRIPT_DIR"

    generate_fio_scenario() {
        local name="$1"
        local rw="$2"
        local direct="$3"
        local bs="$4"
        local numjobs="$5"
        local filename="$6"
        local is_random="$7"

        cat > "$SCENARIOS_DIR/$filename" << EOF
    [global]
    name=$name
    description=$name - direct=$direct bs=$bs numjobs=$numjobs
    rw=$rw
    bs=$bs
    ioengine=libaio
    iodepth=1
    direct=$direct
    numjobs=$numjobs
    runtime=60
    time_based=1
    directory=/data
    size=1G
    group_reporting=1
    EOF

        if [ "$is_random" = "true" ]; then
            echo "norandommap=1" >> "$SCENARIOS_DIR/$filename"
        fi

        echo "" >> "$SCENARIOS_DIR/$filename"
        echo "[job1]" >> "$SCENARIOS_DIR/$filename"
    }

    # Generate all 96 scenarios
    for direct in 0 1; do
        for bs in 128k 1m 4m; do
            for numjobs in 1 8 16 32; do
                fname_base="${direct}d_${bs}_${numjobs}j"

                generate_fio_scenario "rand_read_$fname_base" "randread" "$direct" "$bs" "$numjobs" "rand_read_$fname_base.fio" "true"
                generate_fio_scenario "rand_write_$fname_base" "randwrite" "$direct" "$bs" "$numjobs" "rand_write_$fname_base.fio" "true"
                generate_fio_scenario "seq_read_$fname_base" "read" "$direct" "$bs" "$numjobs" "seq_read_$fname_base.fio" "false"
                generate_fio_scenario "seq_write_$fname_base" "write" "$direct" "$bs" "$numjobs" "seq_write_$fname_base.fio" "false"
            done
        done
    done

    echo "Generated 96 fio scenario files in $SCENARIOS_DIR"
  </action>
