# Base patterns:

# Sequential pattern (rw=read or rw=write):
# [global]
# name=<name>
# description=<description>
# rw=<read|write>
# bs=<blocksize>
# ioengine=libaio
# iodepth=4
# direct=<0|1>
# numjobs=<1|8|16|32>
# runtime=60
# time_based=1
# directory=/data
# size=1G
# group_reporting=1
# [job1]

# Random pattern (rw=randread or rw=randwrite):
# [global]
# name=<name>
# description=<description>
# rw=<randread|randwrite>
# bs=<blocksize>
# ioengine=libaio
# iodepth=32
# direct=<0|1>
# numjobs=<1|8|16|32>
# runtime=60
# time_based=1
# directory=/data
# size=1G
# norandommap=1
# group_reporting=1
# [job1]

# Naming convention: {rw}_{direct}d_{bs}_{numjobs}j.fio
# Examples: rand_read_0d_128k_1j.fio, seq_write_1d_4m_32j.fio

# For bs in filenames use: 128k, 1m, 4m (lowercase, no space)
</context>

  <action>
    Create a bash script that generates all 96 fio scenario files. The script should be saved at scenarios/fio/gen_fio_scenarios.sh.

    The script will:
    1. Define a function generate_seq_scenario() for sequential read/write scenarios
    2. Define a function generate_rand_scenario() for random read/write scenarios
    3. Loop through all combinations:
       - rw_types: rand_read rand_write seq_read seq_write
       - direct: 0 1
       - bs_values: 128k 1m 4m
       - numjobs: 1 8 16 32
    4. Generate each file with proper fio syntax

    For sequential scenarios:
    - rw=read (for seq_read) or rw=write (for seq_write)
    - iodepth=4
    - No norandommap

    For random scenarios:
    - rw=randread (for rand_read) or rw=randwrite (for rand_write)
    - iodepth=32
    - Include norandommap=1

    All scenarios use:
    - ioengine=libaio
    - runtime=60
    - time_based=1
    - directory=/data
    - size=1G
    - group_reporting=1

    The generated script should be executable (chmod +x).

    Template for the generated script:
    #!/bin/bash
    # Generator script for fio scenarios
    # Creates 96 scenario files: 4 types x 2 direct x 3 bs x 4 numjobs

    SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
    SCENARIOS_DIR="$SCRIPT_DIR"

    generate_seq_scenario() {
        local name="$1"
        local rw="$2"
        local direct="$3"
        local bs="$4"
        local numjobs="$5"
        local filename="$6"

        cat > "$SCENARIOS_DIR/$filename" << EOF
    [global]
    name=$name
    description=Sequential $([ "$rw" = "read" ] && echo "read" || echo "write") test - direct=$direct bs=$bs numjobs=$numjobs
    rw=$rw
    bs=$bs
    ioengine=libaio
    iodepth=4
    direct=$direct
    numjobs=$numjobs
    runtime=60
    time_based=1
    directory=/data
    size=1G
    group_reporting=1

    [job1]
    EOF
    }

    generate_rand_scenario() {
        local name="$1"
        local rw="$2"
        local direct="$3"
        local bs="$4"
        local numjobs="$5"
        local filename="$6"

        cat > "$SCENARIOS_DIR/$filename" << EOF
    [global]
    name=$name
    description=Random $([ "$rw" = "randread" ] && echo "read" || echo "write") test - direct=$direct bs=$bs numjobs=$numjobs
    rw=$rw
    bs=$bs
    ioengine=libaio
    iodepth=32
    direct=$direct
    numjobs=$numjobs
    runtime=60
    time_based=1
    directory=/data
    size=1G
    norandommap=1
    group_reporting=1

    [job1]
    EOF
    }

    # Generate all 96 scenarios
    for direct in 0 1; do
        for bs in 128k 1m 4m; do
            for numjobs in 1 8 16 32; do
                # rand_read
                generate_rand_scenario "rand_read_${direct}d_${bs}_${numjobs}j" "randread" "$direct" "$bs" "$numjobs" "rand_read_${direct}d_${bs}_${numjobs}j.fio"

                # rand_write
                generate_rand_scenario "rand_write_${direct}d_${bs}_${numjobs}j" "randwrite" "$direct" "$bs" "$numjobs" "rand_write_${direct}d_${bs}_${numjobs}j.fio"

                # seq_read
                generate_seq_scenario "seq_read_${direct}d_${bs}_${numjobs}j" "read" "$direct" "$bs" "$numjobs" "seq_read_${direct}d_${bs}_${numjobs}j.fio"

                # seq_write
                generate_seq_scenario "seq_write_${direct}d_${bs}_${numjobs}j" "write" "$direct" "$bs" "$numjobs" "seq_write_${direct}d_${bs}_${numjobs}j.fio"
            done
        done
    done

    echo "Generated 96 fio scenario files in $SCENARIOS_DIR"
  </action>
